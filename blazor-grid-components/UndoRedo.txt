@code {
    // State Management
    private SfGrid<OrderDetails> GridRef;
    private SfToast ToastRef;
    private List<OrderDetails> OrderData { get; set; }
    private bool EnableUndoRedo = true;
    private int UndoRedoLimit = 10;
    private int ActionCount = 0;

    protected override void OnInitialized()
    {
        OrderData = OrderDetails.GetAllRecords();
    }

    // Toggle Undo/Redo Feature
    private async Task ToggleUndoRedo()
    {
        EnableUndoRedo = !EnableUndoRedo;
        string status = EnableUndoRedo ? "Undo/Redo ENABLED" : "Undo/Redo DISABLED";
        await ShowToast("Configuration Updated", status);
    }

    // Handle Keyboard Navigation (Ctrl+Z for Undo, Ctrl+Y for Redo)
    private async Task HandleKeyDown(KeyboardEventArgs e)
    {
        if (e.CtrlKey && (e.Key == "z" || e.Key == "Z"))
        {
            await GridRef.UndoAsync();
            await ShowToast("Undo", "Action undone successfully!");
        }
        else if (e.CtrlKey && (e.Key == "y" || e.Key == "Y"))
        {
            await GridRef.RedoAsync();
            await ShowToast("Redo", "Action redone successfully!");
        }
    }

    // Handle Grid Actions (Update/Cancel)
    private async Task OnActionComplete(ActionEventArgs<OrderDetails> args)
    {
        if (args.RequestType == "Save")
        {
            ActionCount++;
            ShowToast("Changes Saved", $"Batch changes saved successfully! Actions tracked: {ActionCount}");
        }
        else if (args.RequestType == "Cancel")
        {
            ActionCount = 0;
            ShowToast("Changes Cancelled", "All pending changes have been cancelled. Stack cleared.");
        }
    }

    // Show Toast Notification
    private async Task ShowToast(string title, string content)
    {
        if (ToastRef != null)
        {
            await ToastRef.ShowAsync(new ToastModel
            {
                Title = title,
                Content = content,
                Icon = "e-icons e-message"
            });
        }
    }
